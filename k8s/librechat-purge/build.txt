# librechat-purge — rebuild, push and deploy
# Run from the repo root. Each block can be pasted as a whole.

# ── 0. Variables (edit NS) ───────────────────────────────────────────────────
NS=<your-namespace>
IMAGE=nodeenergy.azurecr.io/librechat-purge
TAG=$(git rev-parse --short HEAD)

# ── 1. Build and push ────────────────────────────────────────────────────────
az acr login --name nodeenergy
docker build --platform linux/amd64 -t $IMAGE:latest -t $IMAGE:$TAG k8s/librechat-purge
docker push $IMAGE:latest
docker push $IMAGE:$TAG

# ── 2. Check PVC name and LibreChat pod label ────────────────────────────────
kubectl -n $NS get pvc librechat-librechat-images
kubectl -n $NS get pods -l app.kubernetes.io/name=librechat-librechat -o wide
# Is /app/uploads (PDFs, documents) on a PVC? If yes, enable the uploads block in cronjob.yaml with that claimName
kubectl -n $NS get deploy librechat-librechat -o jsonpath='{range .spec.template.spec.containers[0].volumeMounts[*]}{.mountPath}{"  ->  "}{.name}{"\n"}{end}{range .spec.template.spec.volumes[*]}{.name}{"  =  "}{.persistentVolumeClaim.claimName}{"\n"}{end}'

# ── 3. Apply the CronJob ─────────────────────────────────────────────────────
kubectl -n $NS apply -f k8s/librechat-purge/cronjob.yaml

# ── 4. Dry run (lists only, deletes nothing) ─────────────────────────────────
kubectl -n $NS create job librechat-purge-dryrun --from=cronjob/librechat-purge --dry-run=client -o yaml | sed -e '/- --delete$/d' -e '/- --yes$/d' | kubectl -n $NS apply -f -
kubectl -n $NS get pods -l job-name=librechat-purge-dryrun -o wide
kubectl -n $NS logs -f job/librechat-purge-dryrun
kubectl -n $NS delete job librechat-purge-dryrun

# ── 5. Real run now (optional, otherwise it runs at 00:00) ───────────────────
kubectl -n $NS create job librechat-purge-manual --from=cronjob/librechat-purge
kubectl -n $NS logs -f job/librechat-purge-manual
kubectl -n $NS delete job librechat-purge-manual

# ── Troubleshooting ──────────────────────────────────────────────────────────
# Pod stuck in Pending -> affinity label or PVC name is wrong:
kubectl -n $NS describe pod -l job-name=librechat-purge-dryrun
# "Permission denied" on delete -> files not owned by uid 1000:
kubectl -n $NS exec deploy/librechat-librechat -- ls -ln /app/client/public/images
